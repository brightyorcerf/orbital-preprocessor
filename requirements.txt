# OSP — Python dependency manifest
# Pin to compatible version ranges, not hard-pinned exact versions,
# so CI and development machines remain flexible while staying deterministic
# enough for reproducible runs.
#
# Install:  pip install -r requirements.txt
# Docker:   handled in Dockerfile (onnxruntime-gpu replaces onnxruntime below)

# ── Core scientific stack ─────────────────────────────────────────────────────
numpy>=1.24,<2.0
opencv-python>=4.8,<5.0        # cv2 — used in synthetic_bands, preprocess, engine

# ── Deep learning ─────────────────────────────────────────────────────────────
torch>=2.4,<3.0                # PyTorch — model surgery + dataset loader
                               # 2.4 is the floor for torch.amp.GradScaler("cuda", ...)
ultralytics>=8.1,<9.0          # YOLOv8 training + export API

# ── ONNX inference (CPU; Dockerfile overrides with onnxruntime-gpu) ──────────
onnxruntime>=1.17,<2.0
onnx>=1.16,<2.0                # required by onnxruntime.quantization (INT8 PTQ)

# ── Protobuf (wire format) ────────────────────────────────────────────────────
# Floor is 5.29, NOT 4.25: inference/osp_pb2.py is gencode emitted by protoc
# 5.29 and opens with ValidateProtobufRuntimeVersion(PUBLIC, 5, 29, 0). A clean
# install that resolved to protobuf 4.x raised at import time, so the advertised
# reproducible-install path was broken for anyone starting from scratch.
protobuf>=5.29,<6.0

# ── Ground dashboard ──────────────────────────────────────────────────────────
streamlit>=1.32,<2.0
# Imported directly by ground/dashboard.py to render the decision audit trail.
# Streamlit pulls it in transitively, but a module that imports a package must
# declare it rather than rely on someone else's dependency staying put.
pandas>=2.0
# Both map tabs (2D ground track, 3D globe) are one ground/globe.py figure
# under two projections. folium and streamlit-folium used to draw the 2D one
# from a second code path and are gone.
plotly>=5.20,<7.0          # validated on 6.9; 5.x also works

# ── LLM reasoning layer (ORION analyst) ──────────────────────────────────────
# Gemini only. The OpenAI-compatible path was absent from every deployment
# manifest, so nothing anyone runs could reach it.
google-generativeai>=0.5,<1.0

# ── GenAI: RAG (rag/ module) ─────────────────────────────────────────────────
# No vector database. The corpus is 14 chunks, so retrieval is a dot product
# against a (14, 384) numpy matrix — which is what a flat inner-product index
# computes anyway, minus the on-disk index and its staleness checks.
sentence-transformers>=2.7,<4.0  # Local embedding model (all-MiniLM-L6-v2)

# ── Real data ingestion (optional — only needed for live Sentinel-2 data) ─────
# rasterio>=1.3,<2.0           # uncomment to enable GeoTIFF loading

# ── Development / testing ─────────────────────────────────────────────────────
pytest>=8.0,<10.0

# ── Orbital mechanics (orbital/) ─────────────────────────────────────────────
# The reference SGP4 implementation. A TLE is a set of mean elements fitted to
# this specific analytical theory, so propagating one with anything else — a
# generic integrator included — gives confidently wrong answers. Pure Python
# with an optional C accelerator; no data files, no network at runtime.
sgp4>=2.23,<3.0
